## 5A — คนที่ 1: `backend/routes/evaluate.js`

📄 **`backend/routes/evaluate.js`**

```js
const router = require('express').Router();
const { query } = require('../db');
const { role } = require('../auth');
const {
  check,
  text,
  ok,
  route,
  getPeriod,
  getParticipant,
  items,
  open,
  scoreValid,
  percentage,
} = require('../helpers');
const { upload, saveFile } = require('../uploads');
async function own(id, user, editing = false) {
  const rows = await query('SELECT * FROM assignments WHERE id=?', [id]);
  check(rows.length, 'ไม่พบงาน', 404);
  const a = rows[0];
  check(a.evaluator_id === user.id, 'ไม่ใช่งานที่มอบหมายให้คุณ', 403);
  const person = await getParticipant(a.participant_id),
    period = await getPeriod(person.period_id);
  if (editing) {
    open(period);
    check(person.submitted, 'ผู้รับยังไม่ส่ง');
    check(!a.submitted, 'กรรมการส่งแล้ว แก้ไม่ได้');
  }
  return { assignment: a, person, period };
}
// GET /api/my-assignments — งานที่กรรมการได้รับมอบหมาย (evaluator)
router.get(
  '/my-assignments',
  role('evaluator'),
  route(async (req, res) => {
    ok(
      res,
      await query(
        'SELECT a.id,a.submitted,a.committee_role,u.full_name,r.name,p.submitted AS self_submitted FROM assignments a JOIN participants p ON p.id=a.participant_id JOIN users u ON u.id=p.user_id JOIN periods r ON r.id=p.period_id WHERE a.evaluator_id=? ORDER BY a.id DESC',
        [req.user.id],
      ),
    );
  }),
);
// GET /api/evaluate/:id — ข้อมูลสำหรับให้คะแนน path: id = assignment id (evaluator)
router.get(
  '/evaluate/:id',
  role('evaluator'),
  route(async (req, res) => {
    const data = await own(req.params.id, req.user);
    const indicators = await items(data.person.period_id);
    const scores = await query('SELECT * FROM scores WHERE assignment_id=?', [data.assignment.id]);
    ok(res, {
      ...data,
      indicators,
      scores,
      self_scores: await query('SELECT * FROM self_scores WHERE participant_id=?', [
        data.person.id,
      ]),
      percentage: percentage(indicators, scores),
    });
  }),
);
// PUT /api/evaluate/:id/scores/:indicatorId — ให้คะแนน path: id, indicatorId body: score (evaluator)
router.put(
  '/evaluate/:id/scores/:indicatorId',
  role('evaluator'),
  route(async (req, res) => {
    const data = await own(req.params.id, req.user, true);
    const item = (await items(data.person.period_id)).find(
      (i) => i.id === Number(req.params.indicatorId),
    );
    check(item, 'ตัวชี้วัดไม่อยู่ในรอบ', 404);
    check(scoreValid(item, req.body.score), 'คะแนนไม่ถูกชนิด');
    await query(
      'INSERT INTO scores(assignment_id,indicator_id,score) VALUES(?,?,?) ON DUPLICATE KEY UPDATE score=VALUES(score)',
      [data.assignment.id, item.id, req.body.score],
    );
    ok(res);
  }),
);
// PUT /api/evaluate/:id/comment — บันทึกความเห็น body: comment (evaluator)
router.put(
  '/evaluate/:id/comment',
  role('evaluator'),
  route(async (req, res) => {
    await own(req.params.id, req.user, true);
    await query('UPDATE assignments SET comment=? WHERE id=?', [
      text(req.body.comment, 3000),
      req.params.id,
    ]);
    ok(res);
  }),
);
// POST /api/evaluate/:id/signature — อัปโหลดลายเซ็น form-data: file PNG/JPEG (evaluator)
router.post(
  '/evaluate/:id/signature',
  role('evaluator'),
  upload,
  route(async (req, res) => {
    await own(req.params.id, req.user, true);
    await query('UPDATE assignments SET signature=? WHERE id=?', [
      saveFile(req.file, true),
      req.params.id,
    ]);
    ok(res);
  }),
);
// POST /api/evaluate/:id/submit — ยืนยันและส่งผล path: id (evaluator)
router.post(
  '/evaluate/:id/submit',
  role('evaluator'),
  route(async (req, res) => {
    const data = await own(req.params.id, req.user, true),
      a = data.assignment;
    const score = percentage(
      await items(data.person.period_id),
      await query('SELECT * FROM scores WHERE assignment_id=?', [a.id]),
    );
    check(score !== null && a.comment && a.signature, 'คะแนน ความเห็น หรือลายเซ็นยังไม่ครบ');
    await query('UPDATE assignments SET submitted=1 WHERE id=?', [a.id]);
    ok(res);
  }),
);
module.exports = router;
```

**เข้าใจโค้ด**

- `own()` ตรวจว่างานนี้มอบหมายให้กรรมการคนที่ login จริง (`evaluator_id === user.id`) ไม่ใช่ตอบ 403
- จะแก้คะแนนได้ต้อง: รอบเปิดอยู่ ผู้รับส่งแล้ว และกรรมการยังไม่ส่ง
- `GET /evaluate/:id` ส่งทั้งตัวชี้วัด คะแนนกรรมการ คะแนนตนเองของผู้รับ และร้อยละรวม
- ลายเซ็นใช้ `saveFile(file, true)` รับเฉพาะ PNG/JPEG ไม่เกิน 512 KB
- `submit` ตรวจว่าคะแนนครบทุกข้อ มีความเห็น และมีลายเซ็น ก่อนเปลี่ยน `submitted=1`

**เปิดใช้ใน `server.js`:**

```js
app.use('/api', require('./routes/evaluate')); // ขั้น 5
```

## 5B — คนที่ 2: `frontend/src/pages/EvaluatePage.vue`

📄 **`frontend/src/pages/EvaluatePage.vue`**

```vue
<template>
  <section>
    <h1>งานกรรมการ</h1>
    <p v-if="!list.length">ยังไม่มีงานที่ได้รับมอบหมาย</p>
    <div class="actions">
      <v-btn v-for="a in list" :key="a.id" @click="select(a.id)">
        {{ a.name }} / {{ a.full_name }}
      </v-btn>
    </div>
    <template v-if="data">
      <div class="box">
        <h2>{{ data.person.full_name }} · {{ data.period.name }}</h2>
        <p>
          ผู้รับ: {{ data.person.submitted ? 'ส่งแล้ว' : 'ยังไม่ส่ง' }} · กรรมการ:
          {{ data.assignment.submitted ? 'ส่งแล้ว' : 'ยังไม่ส่ง' }}
        </p>
        <p class="big-number">
          {{ data.percentage === null ? 'ยังบันทึกคะแนนไม่ครบ' : data.percentage + '%' }}
        </p>
        <p v-if="!editable">
          ขณะนี้อ่านได้อย่างเดียว: ตรวจสถานะผู้รับ การส่งของกรรมการ และวันเปิดรอบ
        </p>
      </div>
      <article class="box" v-for="i in data.indicators" :key="i.id">
        <h2>{{ i.topic }} / {{ i.name }}</h2>
        <p>{{ i.detail }} · น้ำหนัก {{ i.weight }}</p>
        <p>คะแนนตนเอง: {{ self(i).score ?? 'ยังไม่กรอก' }} · {{ self(i).description }}</p>
        <div class="actions">
          <v-btn v-if="self(i).evidence_file" @click="evidence(i)">เปิดหลักฐาน</v-btn>
          <a
            v-if="self(i).evidence_url"
            :href="self(i).evidence_url"
            target="_blank"
            rel="noopener"
          >
            เปิด URL หลักฐาน
          </a>
        </div>
        <score-field
          v-model="values[i.id]"
          :type="i.score_type"
          :disabled="!editable"
        ></score-field>
        <v-btn v-if="editable" color="primary" @click="save(i)">บันทึกคะแนนข้อนี้</v-btn>
      </article>
      <div class="box">
        <h2>ตรวจทานคะแนนที่บันทึก</h2>
        <v-table class="table-wrap">
          <thead>
            <tr>
              <th>ตัวชี้วัด</th>
              <th>น้ำหนัก</th>
              <th>คะแนนกรรมการ</th>
            </tr>
          </thead>
          <tbody>
            <tr v-for="i in data.indicators">
              <td>{{ i.name }}</td>
              <td>{{ i.weight }}</td>
              <td>{{ savedScore(i) ?? 'ยังไม่บันทึก' }}</td>
            </tr>
          </tbody>
        </v-table>
        <v-textarea
          v-model="comment"
          maxlength="3000"
          :disabled="!editable"
          label="ความเห็นภาพรวม"
        />
        <v-btn v-if="editable" @click="saveComment">บันทึกความเห็น</v-btn>
        <v-file-input
          v-if="editable"
          accept=".png,.jpg,.jpeg"
          v-model="file"
          label="ภาพลายเซ็น PNG / JPEG ไม่เกิน 512 KB"
        />
        <v-btn v-if="editable" @click="upload">บันทึกลายเซ็น</v-btn>
        <img v-if="signature" :src="signature" class="signature" alt="ลายเซ็นกรรมการ" />
        <div class="actions">
          <v-btn
            v-if="editable"
            color="primary"
            :disabled="
              data.percentage === null || !data.assignment.comment || !data.assignment.signature
            "
            @click="submit"
          >
            ยืนยันและส่งผล
          </v-btn>
        </div>
      </div>
    </template>
  </section>
</template>

<script>
import { api, mayEdit, fileBlob, openFile } from '../api';
import ScoreField from '../components/ScoreField.vue';
export default {
  components: { ScoreField },
  data() {
    return {
      list: [],
      data: null,
      values: {},
      comment: '',
      file: null,
      signature: '',
    };
  },
  async mounted() {
    this.list = await api('/my-assignments');
  },
  computed: {
    editable() {
      return (
        this.data &&
        !this.data.assignment.submitted &&
        this.data.person.submitted &&
        mayEdit(this.data.period)
      );
    },
  },
  methods: {
    async select(id) {
      this.data = await api('/evaluate/' + id);
      this.values = {};
      this.file = null;
      for (const i of this.data.indicators) {
        const s = this.data.scores.find((s) => s.indicator_id === i.id);
        this.values[i.id] = s ? s.score : null;
      }
      this.comment = this.data.assignment.comment || '';
      await this.showSignature();
    },
    async refresh() {
      this.data = await api('/evaluate/' + this.data.assignment.id);
    },
    savedScore(i) {
      const s = this.data.scores.find((s) => s.indicator_id === i.id);
      return s ? s.score : null;
    },
    self(i) {
      return this.data.self_scores.find((s) => s.indicator_id === i.id) || {};
    },
    async save(i) {
      await api('/evaluate/' + this.data.assignment.id + '/scores/' + i.id, 'PUT', {
        score: this.values[i.id],
      });
      await this.refresh();
      this.$root.message = 'บันทึกคะแนนแล้ว';
    },
    async saveComment() {
      await api('/evaluate/' + this.data.assignment.id + '/comment', 'PUT', {
        comment: this.comment,
      });
      await this.refresh();
      this.$root.message = 'บันทึกความเห็นแล้ว';
    },
    async showSignature() {
      this.signature = this.data.assignment.signature
        ? await fileBlob('/signatures/' + this.data.assignment.id)
        : '';
    },
    async upload() {
      if (!this.file) throw Error('เลือกภาพลายเซ็นก่อน');
      const f = new FormData();
      f.append('file', Array.isArray(this.file) ? this.file[0] : this.file);
      await api('/evaluate/' + this.data.assignment.id + '/signature', 'POST', f);
      await this.refresh();
      await this.showSignature();
      this.$root.message = 'แนบลายเซ็นแล้ว';
    },
    async evidence(i) {
      await openFile('/files/' + this.self(i).id);
    },
    async submit() {
      if (confirm('ส่งแล้วแก้ไม่ได้ ยืนยัน?')) {
        await api('/evaluate/' + this.data.assignment.id + '/submit', 'POST');
        await this.refresh();
        this.$root.message = 'ส่งผลกรรมการแล้ว';
      }
    },
  },
};
</script>
```

**เข้าใจโค้ด**

- `values` คือคะแนนที่เลือกอยู่ ตารางตรวจทานแสดงคะแนนที่ **บันทึกแล้ว** จากฐาน (`savedScore`) ถ้ายังขึ้น "ยังไม่บันทึก" ให้กดบันทึกข้อนั้น
- ลายเซ็นโหลดผ่าน `fileBlob()` เพราะต้องแนบ token จะใช้ `<img src="/api/...">` ตรง ๆ ไม่ได้
- ปุ่ม "ยืนยันและส่งผล" เปิดเมื่อคะแนนครบ มีความเห็น และมีลายเซ็น

## ✅ จุดเช็กขั้น 5 (หน้าเว็บ ด้วย eval01)

เตรียมภาพลายเซ็น: เปิด Paint เซ็นชื่อ บันทึกเป็น `sign.png`

1. login `eval01` / `Eval1234!` → เปลี่ยนรหัสเป็น `Eval1234!x`
2. เมนู **งานกรรมการ** → กดปุ่ม "ชื่อรอบ / ผู้รับหนึ่ง" (5.3.1)
3. เห็นหัวข้อ ตัวชี้วัด รายละเอียด และ "คะแนนตนเอง: 3 · ..." (5.3.2, 5.3.3) กด "เปิดหลักฐาน" เปิดไฟล์ได้
