### `backend/uploads.js` — รับไฟล์ ตรวจชนิดจริง และบันทึก

📄 **`backend/uploads.js`**

```js
const multer = require('multer');
const fs = require('node:fs');
const path = require('node:path');
const { randomUUID } = require('node:crypto');
const { check } = require('./helpers');
const folder = process.env.UPLOAD_DIR || path.join(__dirname, 'uploads');
fs.mkdirSync(folder, { recursive: true });
const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 5 * 1024 * 1024, files: 1 },
}).single('file');
function saveFile(file, signature = false) {
  check(file, 'เลือกไฟล์ก่อน');
  const b = file.buffer;
  let extension = '';
  if (b.subarray(0, 8).equals(Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]))) extension = '.png';
  if (b[0] === 255 && b[1] === 216 && b[2] === 255) extension = '.jpg';
  if (b.subarray(0, 5).toString() === '%PDF-' && !signature) extension = '.pdf';
  check(extension, signature ? 'ลายเซ็นต้องเป็น PNG หรือ JPEG' : 'รับเฉพาะ PDF, PNG หรือ JPEG');
  if (signature) check(file.size <= 512 * 1024, 'ลายเซ็นต้องไม่เกิน 512 KB');
  const name = randomUUID() + extension;
  fs.writeFileSync(path.join(folder, name), b);
  return name;
}
module.exports = { upload, saveFile, folder };
```

- `multer.memoryStorage()` รับไฟล์ไว้ในหน่วยความจำก่อน จำกัด 5 MB
- ตรวจ **ไบต์แรกของไฟล์** (magic number) ไม่เชื่อนามสกุล: PNG เริ่มด้วย `89 50 4E 47`, JPEG `FF D8 FF`, PDF `%PDF-` กันคนเปลี่ยนนามสกุลไฟล์อันตราย
- ตั้งชื่อไฟล์ใหม่ด้วย `randomUUID()` กันชื่อซ้ำและกันชื่อไฟล์แปลก ๆ
- ไฟล์ไปอยู่ที่ `backend/uploads/` ในเครื่องเรา (เพราะ mount โฟลเดอร์ไว้)

### `backend/routes/self.js` — API ของผู้รับการประเมิน

📄 **`backend/routes/self.js`**

```js
const router = require('express').Router();
const { query } = require('../db');
const { role } = require('../auth');
const {
  check,
  text,
  ok,
  route,
  getPeriod,
  getParticipant,
  items,
  open,
  scoreValid,
  complete,
} = require('../helpers');
const { upload, saveFile } = require('../uploads');
async function own(id, user, editing = false) {
  const person = await getParticipant(id);
  check(person.user_id === user.id, 'ไม่ใช่งานของคุณ', 403);
  if (editing) {
    open(await getPeriod(person.period_id));
    check(!person.submitted, 'ส่งแล้ว แก้ไม่ได้');
  }
  return person;
}
// GET /api/my-periods — รอบของผู้รับที่ login อยู่ (evaluatee)
router.get(
  '/my-periods',
  role('evaluatee'),
  route(async (req, res) => {
    ok(
      res,
      await query(
        'SELECT p.id,p.submitted,r.name,r.status,r.start_date,r.end_date,r.published FROM participants p JOIN periods r ON r.id=p.period_id WHERE p.user_id=? ORDER BY p.id DESC',
        [req.user.id],
      ),
    );
  }),
);
// GET /api/self/:id — ข้อมูลประเมินตนเอง path: id = participant id (evaluatee)
router.get(
  '/self/:id',
  role('evaluatee'),
  route(async (req, res) => {
    const person = await own(req.params.id, req.user);
    ok(res, {
      person,
      period: await getPeriod(person.period_id),
      indicators: await items(person.period_id),
      scores: await query('SELECT * FROM self_scores WHERE participant_id=?', [person.id]),
    });
  }),
);
// PUT /api/self/:id/scores/:indicatorId — บันทึกคะแนนตนเอง path: id, indicatorId body: score, description, evidence_url (evaluatee)
router.put(
  '/self/:id/scores/:indicatorId',
  role('evaluatee'),
  route(async (req, res) => {
    const person = await own(req.params.id, req.user, true);
    const item = (await items(person.period_id)).find(
      (i) => i.id === Number(req.params.indicatorId),
    );
    check(item, 'ตัวชี้วัดไม่อยู่ในรอบ', 404);
    const b = req.body;
    check(scoreValid(item, b.score), 'เลือกคะแนนให้ถูกชนิด');
    const url = text(b.evidence_url, 1000, false);
    if (url) {
      let parsed;
      try {
        parsed = new URL(url);
      } catch {}
      check(
        parsed && ['http:', 'https:'].includes(parsed.protocol),
        'URL ต้องขึ้นต้น http:// หรือ https://',
      );
    }
    await query(
      'INSERT INTO self_scores(participant_id,indicator_id,score,description,evidence_url) VALUES(?,?,?,?,?) ON DUPLICATE KEY UPDATE score=VALUES(score),description=VALUES(description),evidence_url=VALUES(evidence_url)',
      [
        person.id,
        item.id,
        b.score,
        text(b.description, 3000, false),
        item.evidence_type === 'url' ? url : '',
      ],
    );
    ok(res);
  }),
);
// POST /api/self/:id/files/:indicatorId — อัปโหลดไฟล์หลักฐาน form-data: file (evaluatee)
router.post(
  '/self/:id/files/:indicatorId',
  role('evaluatee'),
  upload,
  route(async (req, res) => {
    const person = await own(req.params.id, req.user, true);
    const item = (await items(person.period_id)).find(
      (i) => i.id === Number(req.params.indicatorId),
    );
    check(item && item.evidence_type === 'file', 'ข้อนี้ไม่รับไฟล์');
    const rows = await query(
      'SELECT id FROM self_scores WHERE participant_id=? AND indicator_id=?',
      [person.id, item.id],
    );
    check(rows.length, 'บันทึกคะแนนข้อนี้ก่อนแนบไฟล์');
    await query('UPDATE self_scores SET evidence_file=? WHERE id=?', [
      saveFile(req.file),
      rows[0].id,
    ]);
    ok(res);
  }),
);
// POST /api/self/:id/submit — ส่งการประเมินตนเอง path: id (evaluatee)
router.post(
  '/self/:id/submit',
  role('evaluatee'),
  route(async (req, res) => {
    const person = await own(req.params.id, req.user, true);
    const list = await items(person.period_id),
      rows = await query('SELECT * FROM self_scores WHERE participant_id=?', [person.id]);
    check(
      list.length &&
        list.every((i) =>
          complete(
            i,
            rows.find((s) => s.indicator_id === i.id),
          ),
        ),
      'คะแนน รายละเอียด หรือหลักฐานยังไม่ครบ',
    );
    await query('UPDATE participants SET submitted=1 WHERE id=?', [person.id]);
    ok(res);
  }),
);
module.exports = router;
```

- `own()` ตรวจว่าเป็นงานของคนที่ login อยู่จริง และถ้าจะแก้ ต้องอยู่ในช่วงวันที่รอบเปิด และยังไม่ส่ง
- `ON DUPLICATE KEY UPDATE` บันทึกซ้ำข้อเดิม = แก้ไข ไม่เพิ่มแถวใหม่
- `POST /self/:id/files/:indicatorId` ใช้ `upload` (multer) รับช่อง `file` แบบ form-data (เกณฑ์ 6.3)

### `backend/routes/files.js` — เปิดไฟล์หลักฐาน/ลายเซ็นแบบตรวจสิทธิ์

📄 **`backend/routes/files.js`**

```js
const router = require('express').Router();
const path = require('node:path');
const { query } = require('../db');
const { check, route } = require('../helpers');
const { folder } = require('../uploads');
// เปิดไฟล์ผ่าน API ที่ตรวจสิทธิ์ ไม่เปิด uploads เป็น public
// GET /api/files/:id — เปิดไฟล์หลักฐาน (ตรวจสิทธิ์ก่อน) path: id = self_scores id
router.get(
  '/files/:id',
  route(async (req, res) => {
    const rows = await query(
      'SELECT s.*,p.user_id FROM self_scores s JOIN participants p ON p.id=s.participant_id WHERE s.id=?',
      [req.params.id],
    );
    check(rows.length, 'ไม่พบหลักฐาน', 404);
    const row = rows[0],
      assigned = await query(
        'SELECT id FROM assignments WHERE participant_id=? AND evaluator_id=?',
        [row.participant_id, req.user.id],
      );
    check(
      req.user.role === 'admin' || row.user_id === req.user.id || assigned.length,
      'ไม่มีสิทธิ์เปิดไฟล์',
      403,
    );
    check(row.evidence_file, 'ยังไม่มีไฟล์', 404);
    res.sendFile(path.join(folder, row.evidence_file));
  }),
);
// GET /api/signatures/:id — เปิดภาพลายเซ็น path: id = assignment id
router.get(
  '/signatures/:id',
  route(async (req, res) => {
    const rows = await query(
      'SELECT a.*,p.user_id,r.published FROM assignments a JOIN participants p ON p.id=a.participant_id JOIN periods r ON r.id=p.period_id WHERE a.id=?',
      [req.params.id],
    );
    check(rows.length, 'ไม่พบงาน', 404);
    const a = rows[0];
    check(
      req.user.role === 'admin' ||
        a.evaluator_id === req.user.id ||
        (a.user_id === req.user.id && a.published),
      'ไม่มีสิทธิ์',
      403,
    );
    check(a.signature, 'ยังไม่มีลายเซ็น', 404);
    res.sendFile(path.join(folder, a.signature));
  }),
);
module.exports = router;
```

ไม่เปิดโฟลเดอร์ `uploads` ให้ใครก็โหลดได้ ต้องผ่าน API ที่ตรวจว่าเป็นเจ้าของ กรรมการที่ได้รับมอบหมาย หรือ admin เท่านั้น

```vue
<template>
  <v-select
    label="คะแนน"
    :model-value="modelValue"
    :items="options"
    :disabled="disabled"
    item-title="title"
    item-value="value"
    @update:model-value="$emit('update:modelValue', $event)"
  />
</template>
<script>
export default {
  props: ['modelValue', 'type', 'disabled'],
  emits: ['update:modelValue'],
  computed: {
    options() {
      if (this.type === 'boolean')
        return [
          { title: 'ไม่มี (0)', value: 0 },
          { title: 'มี (1)', value: 1 },
        ];
      return [1, 2, 3, 4].map((value) => ({ title: String(value), value }));
    },
  },
};
</script>
```

### `frontend/src/pages/SelfPage.vue` — หน้าประเมินตนเอง

📄 **`frontend/src/pages/SelfPage.vue`**

```vue
<template>
  <section>
    <h1>ประเมินตนเอง</h1>
    <p v-if="!list.length">ยังไม่มีรอบที่ได้รับมอบหมาย</p>
    <div class="actions">
      <v-btn v-for="p in list" :key="p.id" @click="select(p.id)">{{ p.name }}</v-btn>
    </div>
    <template v-if="data">
      <div class="box">
        <h2>{{ data.period.name }}</h2>
        <p>
          ช่วงวันที่ {{ data.period.start_date }} ถึง {{ data.period.end_date }} ·
          {{ data.period.status }}
        </p>
        <p class="progress">
          ข้อมูลที่บันทึกครบ {{ completed }} / {{ data.indicators.length }} ข้อ ·
          {{ data.person.submitted ? 'ส่งแล้ว' : 'ยังไม่ส่ง' }}
        </p>
        <p v-if="!editable">ขณะนี้แก้ไขไม่ได้: ส่งแล้ว หรือรอบยังไม่เปิด/พ้นกำหนด</p>
        <v-btn v-if="data.period.published" @click="$emit('result', data.person.id)">
          ดูผล / PDF
        </v-btn>
      </div>
      <article class="box" v-for="i in data.indicators" :key="i.id">
        <h2>{{ i.topic }} / {{ i.name }}</h2>
        <p>{{ i.detail }} · น้ำหนัก {{ i.weight }}</p>
        <p>
          หลักฐาน: {{ i.evidence_type }} {{ i.evidence_required ? '(บังคับ)' : '(ไม่บังคับ)' }} ·
          {{ ready(i) ? 'บันทึกครบแล้ว' : 'ยังไม่ครบ' }}
        </p>
        <score-field
          v-model="rows[i.id].score"
          :type="i.score_type"
          :disabled="!editable"
        ></score-field>
        <v-textarea
          v-model="rows[i.id].description"
          maxlength="3000"
          :disabled="!editable"
          label="รายละเอียดผลงาน"
        />
        <v-text-field
          v-if="i.evidence_type === 'url'"
          v-model="rows[i.id].evidence_url"
          type="url"
          maxlength="1000"
          :disabled="!editable"
          label="URL หลักฐาน"
        />
        <v-btn v-if="editable" color="primary" @click="save(i)">บันทึกข้อนี้</v-btn>
        <template v-if="i.evidence_type === 'file'">
          <v-file-input
            v-if="editable"
            accept=".pdf,.png,.jpg,.jpeg"
            v-model="files[i.id]"
            label="ไฟล์ PDF / PNG / JPEG ไม่เกิน 5 MB"
          />
          <div class="actions">
            <v-btn v-if="editable" @click="upload(i)">แนบไฟล์</v-btn>
            <v-btn v-if="savedRow(i) && savedRow(i).evidence_file" @click="evidence(i)">
              เปิดไฟล์หลักฐาน
            </v-btn>
          </div>
          <p class="muted">บันทึกคะแนนก่อนแนบ หนึ่งข้อต่อหนึ่งไฟล์ แนบใหม่ใช้ไฟล์ล่าสุด</p>
        </template>
        <a
          v-if="savedRow(i) && savedRow(i).evidence_url"
          :href="savedRow(i).evidence_url"
          target="_blank"
          rel="noopener"
        >
          เปิด URL ที่บันทึก
        </a>
      </article>
      <v-btn
        v-if="editable"
        color="primary"
        :disabled="completed !== data.indicators.length"
        @click="submit"
      >
        ส่งการประเมินตนเอง
      </v-btn>
    </template>
  </section>
</template>

<script>
import { api, mayEdit, openFile } from '../api';
import ScoreField from '../components/ScoreField.vue';
export default {
  components: { ScoreField },
  data() {
    return { list: [], data: null, rows: {}, files: {} };
  },
  async mounted() {
    this.list = await api('/my-periods');
  },
  computed: {
    editable() {
      return this.data && !this.data.person.submitted && mayEdit(this.data.period);
    },
    completed() {
      return this.data ? this.data.indicators.filter((i) => this.ready(i)).length : 0;
    },
  },
  methods: {
    async select(id) {
      this.data = await api('/self/' + id);
      this.rows = {};
      this.files = {};
      for (const i of this.data.indicators) {
        const old = this.data.scores.find((s) => s.indicator_id === i.id);
        this.rows[i.id] = {
          score: old ? old.score : null,
          description: old ? old.description : '',
          evidence_url: old ? old.evidence_url : '',
        };
      }
    },
    ready(i) {
      const s = this.data.scores.find((s) => s.indicator_id === i.id);
      return (
        s &&
        s.description.trim() &&
        (!i.evidence_required || (i.evidence_type === 'file' ? s.evidence_file : s.evidence_url))
      );
    },
    savedRow(i) {
      return this.data.scores.find((s) => s.indicator_id === i.id);
    },
    async refresh() {
      this.data = await api('/self/' + this.data.person.id);
    },
    async save(i) {
      await api('/self/' + this.data.person.id + '/scores/' + i.id, 'PUT', this.rows[i.id]);
      await this.refresh();
      this.$root.message = 'บันทึก ' + i.name + ' แล้ว';
    },
    async upload(i) {
      if (!this.files[i.id]) throw Error('เลือกไฟล์ก่อน');
      const form = new FormData();
      form.append('file', Array.isArray(this.files[i.id]) ? this.files[i.id][0] : this.files[i.id]);
      await api('/self/' + this.data.person.id + '/files/' + i.id, 'POST', form);
      await this.refresh();
      this.$root.message = 'แนบไฟล์แล้ว';
    },
    async evidence(i) {
      await openFile('/files/' + this.savedRow(i).id);
    },
    async submit() {
      if (confirm('ส่งแล้วแก้ไม่ได้ ยืนยันส่ง?')) {
        await api('/self/' + this.data.person.id + '/submit', 'POST');
        await this.refresh();
        this.$root.message = 'ส่งการประเมินตนเองแล้ว';
      }
    },
  },
};
</script>
```

**เข้าใจโค้ด**

- `rows` เก็บค่าที่กำลังพิมพ์ของแต่ละข้อ ต้องกด **บันทึกข้อนี้** ทีละข้อ ข้อมูลจึงเข้าฐาน
- ปุ่มส่งกดได้เมื่อทุกข้อบันทึกครบ
- `completed` นับข้อที่ครบ แสดง "ข้อมูลที่บันทึกครบ x / y ข้อ" (ความคืบหน้า 5.2.6)
- อัปโหลดใช้ `FormData` ส่งช่องชื่อ `file`

2. เมนู **ประเมินตนเอง** → กดปุ่มชื่อรอบ

4. ข้อ "มีเว็บไซต์รายวิชา": เลือก "มี (1)" พิมพ์รายละเอียด ใส่ URL `abc` → บันทึก → กล่องแดง "URL ต้องขึ้นต้น http:// หรือ https://" (validate ทำงาน) → แก้เป็น `https://example.com` → บันทึกอีกครั้ง
5. ความคืบหน้าขึ้น **2 / 2 ข้อ** → ปุ่ม **ส่งการประเมินตนเอง** กดได้ → ยืนยัน → "ส่งแล้ว" ช่องทั้งหมดถูกล็อก
