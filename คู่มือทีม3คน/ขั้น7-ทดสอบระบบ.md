## 7.1 Unit Test ฟังก์ชันคำนวณคะแนน (เกณฑ์ 7.5) — คนที่ 1

📄 **`backend/tests/score.test.js`**

```js
const test = require('node:test');
const assert = require('node:assert/strict');
const { percentage, scoreValid } = require('../helpers');
const items = [
  { id: 1, score_type: 'scale4', weight: 60 },
  { id: 2, score_type: 'boolean', weight: 20 },
  { id: 3, score_type: 'scale4', weight: 20 },
];
test('ถ่วงน้ำหนักได้ 85 และ 75 โดย 0 คือคะแนนจริง', () => {
  assert.equal(
    percentage(items, [
      { indicator_id: 1, score: 4 },
      { indicator_id: 2, score: 1 },
      { indicator_id: 3, score: 1 },
    ]),
    85,
  );
  assert.equal(
    percentage(items, [
      { indicator_id: 1, score: 4 },
      { indicator_id: 2, score: 0 },
      { indicator_id: 3, score: 3 },
    ]),
    75,
  );
});
test('คะแนนไม่ครบต้องยังไม่มีผล', () => assert.equal(percentage(items, []), null));
test('ปฏิเสธคะแนนผิดชนิด', () => {
  assert.equal(scoreValid(items[0], 0), false);
  assert.equal(scoreValid(items[1], 4), false);
});
```

- ตัวชี้วัด 3 ข้อ น้ำหนัก 60/20/20 ได้ 4, 1(มี), 1 → (4/4×60 + 1/1×20 + 1/4×20) ÷ 100 = **85%**
- คะแนนไม่ครบต้องได้ `null` (ยังไม่มีผล) และคะแนนผิดชนิดต้องถูกปฏิเสธ

รัน:

```powershell
docker compose exec backend npm test
```

ต้องเห็น `# pass 3` และ `# fail 0`

## 7.2 ทดสอบ API อัตโนมัติ (เกณฑ์ 7.1, 7.2, 7.3, 7.4, 7.6) — คนที่ 1

📄 **`backend/tests/api.test.js`**

```js
// ทดสอบ API จริง (ต้องเปิด docker compose up ไว้ก่อน)
// รัน: docker compose exec backend npm run test:api
const test = require('node:test');
const assert = require('node:assert/strict');
const BASE = 'http://127.0.0.1:' + (process.env.PORT || 3301) + '/api';
async function call(path, method = 'GET', body, token) {
  const headers = { 'Content-Type': 'application/json' };
  if (token) headers.Authorization = 'Bearer ' + token;
  const res = await fetch(BASE + path, {
    method,
    headers,
    body: body ? JSON.stringify(body) : undefined,
  });
  return { status: res.status, body: await res.json() };
}
const username = 'test' + Date.now();
let token;

test('7.4 Validate: รหัสผ่านสั้นกว่า 8 ตัวต้องถูกปฏิเสธ', async () => {
  const r = await call('/register', 'POST', { username, password: '123', full_name: 'ทดสอบ' });
  assert.equal(r.status, 400);
  assert.equal(r.body.status, 'error');
});

test('7.4 Validate: ชื่อว่างต้องถูกปฏิเสธ', async () => {
  const r = await call('/register', 'POST', { username, password: 'Test1234!', full_name: '   ' });
  assert.equal(r.status, 400);
});

test('7.2 Insert: สมัครผู้ใช้ใหม่ได้ (POST)', async () => {
  const r = await call('/register', 'POST', { username, password: 'Test1234!', full_name: 'ทดสอบ' });
  assert.equal(r.status, 200);
  assert.equal(r.body.status, 'success');
});

test('7.1 Login: รหัสผิดต้องได้ 401', async () => {
  const r = await call('/login', 'POST', { username, password: 'ผิดแน่นอน' });
  assert.equal(r.status, 401);
});

test('7.1 Login: รหัสถูกต้องได้ token', async () => {
  const r = await call('/login', 'POST', { username, password: 'Test1234!' });
  assert.equal(r.status, 200);
  assert.ok(r.body.data.token);
  token = r.body.data.token;
});

test('6.10 ไม่มี token ต้องได้ 401', async () => {
  const r = await call('/me');
  assert.equal(r.status, 401);
});

test('7.6 GET /me ได้ข้อมูลตัวเอง', async () => {
  const r = await call('/me', 'GET', null, token);
  assert.equal(r.status, 200);
  assert.equal(r.body.data.username, username);
});

test('7.3 Update: เปลี่ยนรหัสผ่าน แล้วแก้ข้อมูลส่วนตัว (PUT)', async () => {
  const p = await call(
    '/me/password',
    'PUT',
    { current_password: 'Test1234!', new_password: 'New12345!', confirm_password: 'New12345!' },
    token,
  );
  assert.equal(p.status, 200);
  const u = await call('/me', 'PUT', { full_name: 'ชื่อใหม่', department: 'IT', position: 'ครู' }, token);
  assert.equal(u.status, 200);
  const me = await call('/me', 'GET', null, token);
  assert.equal(me.body.data.full_name, 'ชื่อใหม่');
  assert.equal(me.body.data.department, 'IT');
});

test('4.4 เรียก API ที่ไม่มีต้องได้ 404', async () => {
  const r = await call('/ไม่มีจริง', 'GET', null, token);
  assert.equal(r.status, 404);
});
```

รัน (ระบบต้องเปิดอยู่):

```powershell

```

## 7.4 DevTools และ Responsive (เกณฑ์ 7.7, 7.8) — คนที่ 2

1. เปิด http://localhost:5175 กด **F12**
2. แท็บ **Console**: login แล้วกดทุกเมนู ต้องไม่มีบรรทัดสีแดง (คำเตือนสีเหลืองไม่เป็นไร) จับภาพ
3. แท็บ **Network**: เลือก Fetch/XHR กดบันทึกข้อมูลส่วนตัว เห็น `me` สถานะ 200 คลิกดู Response เป็น `{status, message, data}` จับภาพ
4. กด **Ctrl+Shift+M** (Toggle device toolbar) เลือก iPhone 12 Pro → ดูทุกหน้า จับภาพ แล้วเลือก Responsive กว้าง 1280 จับภาพเทียบกัน
5. ถ้าเจอข้อความแดงที่เกิดจากการกดผิดลำดับ (เช่น กล่องแดงที่ระบบตั้งใจแสดง) ไม่ถือเป็น error ของโปรแกรม แต่ถ้าเป็น error ของโค้ด (เช่น `Cannot read properties of undefined`) ให้แจ้งคนเขียนหน้านั้น
